cmake_minimum_required(VERSION 3.25)
project(CodexSync VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 23)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CXS_GUI "Build native Qt6 desktop GUI" ON)
set(CXS_SODIUM_ROOT "" CACHE PATH "Windows libsodium SDK directory")
if(WIN32)
  find_path(SODIUM_INCLUDE sodium.h PATHS "${CXS_SODIUM_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
  find_library(SODIUM_LIB sodium libsodium PATHS "${CXS_SODIUM_ROOT}/x64/Release/v143/dynamic" REQUIRED NO_DEFAULT_PATH)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
  find_package(CURL REQUIRED)
endif()
add_library(cxs_sqlite STATIC third_party/sqlite3.c)
target_include_directories(cxs_sqlite PUBLIC third_party)
target_compile_definitions(cxs_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
add_library(codex_sync SHARED src/core.cpp src/history.cpp src/crypto.cpp src/store.cpp third_party/tinyxml2.cpp)
target_compile_definitions(codex_sync PRIVATE CODEX_SYNC_BUILD)
target_include_directories(codex_sync PUBLIC include src third_party)
target_link_libraries(codex_sync PRIVATE cxs_sqlite)
if(WIN32)
  target_include_directories(codex_sync PUBLIC "${SODIUM_INCLUDE}")
  target_compile_definitions(codex_sync PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(codex_sync PUBLIC "${SODIUM_LIB}" PRIVATE winhttp advapi32)
  add_custom_command(TARGET codex_sync POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CXS_SODIUM_ROOT}/x64/Release/v143/dynamic/libsodium.dll" "$<TARGET_FILE_DIR:codex_sync>")
else()
  target_link_libraries(codex_sync PUBLIC PkgConfig::SODIUM PRIVATE CURL::libcurl dl pthread)
endif()
# CLI/GUI use the same implementation; C entry points are exported for external callers.
set_target_properties(codex_sync PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(codex-sync src/cli.cpp src/api.cpp)
target_link_libraries(codex-sync PRIVATE codex_sync)
if(WIN32)
  target_link_libraries(codex-sync PRIVATE ws2_32 shell32)
endif()
if(CXS_GUI)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Quick QuickControls2 Concurrent Network Svg)
  add_executable(CodexSync WIN32 src/gui_quick.cpp)
  add_executable(cxs-icon-maker EXCLUDE_FROM_ALL scripts/make_icon.cpp)
  target_link_libraries(cxs-icon-maker PRIVATE Qt6::Gui Qt6::Svg)
  if(WIN32)
    enable_language(RC)
    configure_file(packaging/icon.rc.in "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/icon.rc" PROPERTIES
      OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ui/codexsync.ico")
    target_sources(CodexSync PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/icon.rc")
  endif()
  set_target_properties(CodexSync PROPERTIES AUTOMOC ON)
  target_link_libraries(CodexSync PRIVATE codex_sync Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Network Qt6::Svg)
  file(GLOB CXS_ICONS CONFIGURE_DEPENDS "ui/icons/*.svg")
  qt_add_resources(CodexSync cxs_ui PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ui/Main.qml ui/Icon.qml ui/SettingsCard.qml ui/codexsync.png ${CXS_ICONS})
endif()
if(MSVC)
  foreach(t codex_sync codex-sync)
    target_compile_options(${t} PRIVATE /utf-8 /W4 /permissive- /bigobj)
  endforeach()
  if(CXS_GUI)
    target_compile_options(CodexSync PRIVATE /utf-8 /W4 /permissive- /bigobj)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(cxs-tests tests/core_tests.cpp)
  target_link_libraries(cxs-tests PRIVATE codex_sync cxs_sqlite)
  if(MSVC)
    target_compile_options(cxs-tests PRIVATE /utf-8)
  endif()
  add_test(NAME core-contracts COMMAND cxs-tests "${CMAKE_BINARY_DIR}/history-contracts")
endif()
install(TARGETS codex_sync codex-sync RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
if(CXS_GUI)
  install(TARGETS CodexSync RUNTIME DESTINATION bin)
endif()
install(FILES include/codex_sync.h DESTINATION include)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/doc/codex-sync)
install(DIRECTORY third_party/licenses/ DESTINATION share/doc/codex-sync/licenses)
install(FILES ui/icons/LICENSE.txt DESTINATION share/doc/codex-sync/licenses RENAME fluent-icons-MIT.txt)
if(UNIX)
  if(CXS_GUI)
    install(FILES packaging/codex-sync.desktop DESTINATION share/applications)
    install(FILES ui/codexsync.svg DESTINATION share/icons/hicolor/scalable/apps)
  endif()
  set(CPACK_GENERATOR DEB)
  set(CPACK_PACKAGE_NAME codex-sync)
  set(CPACK_PACKAGE_CONTACT "CodexSync local build")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "libsodium23, libcurl4 | libcurl4t64, libqt6widgets6 | libqt6widgets6t64, libqt6concurrent6 | libqt6concurrent6t64, libqt6network6 | libqt6network6t64")
  include(CPack)
endif()
